% 公共坐标轴样式。bar width 与相邻 bar shift 的差值相同，因此组内柱子紧贴。
\pgfplotsset{
  common axis/.style={
    ybar,
    axis lines*=left,
    axis y discontinuity=parallel,
    grid=major,
    xmajorgrids=false,
    grid style={draw=black!18, densely dashed, line width=0.45pt},
    axis line style={draw=black!80, line width=0.8pt},
    tick style={draw=black!80, line width=0.7pt},
    tick label style={font=\sffamily\footnotesize},
    label style={font=\sffamily\normalsize},
    title style={font=\sffamily\normalsize\bfseries, yshift=1pt},
    xticklabel style={font=\sffamily\footnotesize, align=center},
    enlarge x limits=0.11,
    clip=true,
  },
  SO-DS/.style={
    fill=sods,
    draw=black!80,
    line width=0.55pt,
    bar width=10pt,
    bar shift=-15pt,
  },
  RS-DS/.style={
    fill=rsds,
    draw=black!80,
    line width=0.55pt,
    bar width=10pt,
    bar shift=-5pt,
  },
  SO-RF/.style={
    fill=sorf,
    draw=black!80,
    line width=0.55pt,
    bar width=10pt,
    bar shift=5pt,
  },
  RS-RF/.style={
    fill=rsrf,
    draw=black!80,
    line width=0.55pt,
    bar width=10pt,
    bar shift=15pt,
  },
}
